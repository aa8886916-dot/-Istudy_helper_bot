import telebot
import requests
import base64
import os
from flask import Flask
from threading import Thread

# --- إعداد السيرفر للبقاء حياً ---
app = Flask('')
@app.route('/')
def home(): return "I am alive"
def run(): app.run(host='0.0.0.0', port=8080)
def keep_alive(): Thread(target=run).start()

# --- جلب المفاتيح بأمان ---
BOT_TOKEN = os.environ.get('BOT_TOKEN')
OPENROUTER_API_KEY = os.environ.get('OPENROUTER_API_KEY')

bot = telebot.TeleBot(BOT_TOKEN)

# ذاكرة المحادثات (آخر 10 رسائل)
memory = {}

@bot.message_handler(commands=['start'])
def start(message):
    bot.reply_to(message, "هلا 👋\nأنا بوت الدراسة الذكي 📚\nدزلي سؤالك أو صورة من كتابك وأنا أحله إلك فوراً 🔥")

@bot.message_handler(func=lambda message: True, content_types=['text'])
def reply(message):
    try:
        user_id = message.chat.id
        if user_id not in memory: memory[user_id] = []
        
        memory[user_id].append({"role": "user", "content": message.text})
        memory[user_id] = memory[user_id][-10:] # الحفاظ على الذاكرة قصيرة

        response = requests.post(
            "https://openrouter.ai/api/v1/chat/completions",
            headers={
                "Authorization": f"Bearer {OPENROUTER_API_KEY}",
                "Content-Type": "application/json"
            },
            json={
                "model": "google/gemini-flash-1.5", # استخدمنا هنا جمناي لأنه الأفضل للدراسة
                "messages": memory[user_id]
            }
        )
        data = response.json()
        text = data["choices"][0]["message"]["content"]
        
        memory[user_id].append({"role": "assistant", "content": text})
        bot.reply_to(message, text)
    except Exception as e:
        bot.reply_to(message, "أعتذر، حدث ضغط على النظام. حاول مرة ثانية.")

@bot.message_handler(content_types=['photo'])
def handle_photo(message):
    bot.reply_to(message, "جاري تحليل الصورة... انتظر ثواني ⏳")
    try:
        file_info = bot.get_file(message.photo[-1].file_id)
        downloaded_file = bot.download_file(file_info.file_path)
        image_base64 = base64.b64encode(downloaded_file).decode('utf-8')

        response = requests.post(
            "https://openrouter.ai/api/v1/chat/completions",
            headers={
                "Authorization": f"Bearer {OPENROUTER_API_KEY}",
                "Content-Type": "application/json"
            },
            json={
                "model": "google/gemini-flash-1.5", # جمناي فلاش ممتاز جداً بتحليل صور الكتب
                "messages": [
                    {
                        "role": "user",
                        "content": [
                            {"type": "text", "text": "أنت مساعد دراسي محترف. حلل هذه الصورة، إذا كانت سؤالاً فقم بحله بالتفصيل، وإذا كانت شرحاً فقم بتلخيصه."},
                            {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{image_base64}"}}
                        ]
                    }
                ]
            }
        )
        data = response.json()
        text = data["choices"][0]["message"]["content"]
        bot.reply_to(message, text)
    except Exception as e:
        bot.reply_to(message, "ما گدرت أقرأ الصورة بوضوح، جرب تأخذ صورة أقرب.")

if __name__ == "__main__":
    keep_alive() # تشغيل السيرفر
    print("البوت الدراسي انطلق...")
    bot.infinity_polling()
